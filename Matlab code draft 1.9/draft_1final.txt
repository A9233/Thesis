            freq_data(start_idx:start_idx + Nsc/2 - 1) = a(1:Nsc/2, symbol_idx);
            % Map positive frequencies (subcarriers 1 to 90)
            freq_data(center_idx:end_idx) = a(Nsc/2 + 1:Nsc, symbol_idx);

            % Perform IFFT
            time_data = ifft(ifftshift(freq_data), Nfft) * sqrt(Nfft); % Normalize

            cp_data = time_data(end - Ncp + 1:end);  % Last Ncp samples

% Parameters (matched to 3 MHz LTE transmitter)

lambda = c / fc;      % Wavelength

% Multipath channel parameters
Npaths = 4;           % Number of multipath components (distinct AoAs)
theta = [20, 15, 55, 75]; % Angles of arrival (degrees, Npaths values)
%rng(42);              % Seed for reproducibility
gains = (randn(1, Npaths) + 1j * randn(1, Npaths)) / sqrt(2); % Random complex gains (1 x Npaths)

% % Validate Npaths
% if length(theta) ~= Npaths || length(gains) ~= Npaths
%     error('Npaths must match the number of elements in theta and gains');

% Assume tx_signal from transmitter (3840 samples for 1 subframe)
% tx_signal: Time-domain signal of length 3840 samples (2 slots)

% Multipath Channel Model (steering vectors for 8-antenna ULA)
steering_vectors = zeros(Nant, Npaths); % Nant x Npaths
for p = 1:Npaths
    phase_shifts = exp(-1j * 2 * pi * d_lambda * sind(theta(p)) * (0:Nant-1)).';
    steering_vectors(:, p) = phase_shifts / sqrt(Nant); % Normalize
end

% Generate received signal for each antenna
rx_signal = zeros(length(tx_signal), Nant);
for p = 1:Npaths
    signal_path = tx_signal * gains(p);
    for ant = 1:Nant
        rx_signal(:, ant) = rx_signal(:, ant) + signal_path * steering_vectors(ant, p);
    end
end

        symbol_length = Nfft + Ncp;
        ofdm_symbol = rx_signal(sample_idx:sample_idx + symbol_length - 1, :);
        
        % Remove CP
        time_data = ofdm_symbol(Ncp + 1:end, :);  % Nfft x Nant
        
        % Store time-domain data
        symbol_idx = (slot-1) * Nsymb + symb;
        time_symbols(:, symbol_idx, :) = time_data;
        
        % Perform FFT
        freq_data = fftshift(fft(time_data, Nfft, 1), 1) / sqrt(Nfft); % Nfft x Nant
        
        % Extract central 180 subcarriers

        start_idx = center_idx - Nsc / 2;

        % Update sample index

    end
end

sss_corr_freq = zeros(168, 2);
for ant = 1:Nant
    grid_symb = rx_grid(60:121, sss_symbol, ant);
    for nid1 = 0:167
        for sf = 1:2 % Subframe 0 or 5
            corr = abs(sum(grid_symb .* conj(sss_seq(:, nid1 + 1, sf))));
            sss_corr_freq(nid1 + 1, sf) = sss_corr_freq(nid1 + 1, sf) + corr;
        end
    end
end
sss_corr_freq = sum(sss_corr_freq, 3); % Sum across antennas
[~, max_idx] = max(sss_corr_freq(:));
[nid1_est, sf_idx] = ind2sub([168, 2], max_idx);
N_ID_1_est = nid1_est - 1; % 0 to 167
subframe_est = (sf_idx == 1) * 0 + (sf_idx == 2) * 5;

% Calculate PCI
PCI_est = 3 * N_ID_1_est + N_ID_2_est;

% Time-Domain PSS Correlation Over IFFT 256 Samples (Symbol 7)
pss_time_seq = zeros(Nfft, 3);
for idx = 1:3
    freq_data = zeros(Nfft, 1);
    center_idx = Nfft / 2 + 1;
    start_idx = center_idx - 31; % Central 62 subcarriers
    end_idx = center_idx + 30;
    freq_data(start_idx:end_idx) = pss_seq(:, idx); % Map 62 subcarriers
    pss_time_seq(:, idx) = ifft(ifftshift(freq_data), Nfft) * sqrt(Nfft);
end

% Correlate for PSS symbol
pss_corr_time = zeros(Nfft, 3); % 256 x 3
for ant = 1:Nant
    time_data = time_symbols(:, pss_symbol, ant); % 256 samples from symbol 7
    for nid2 = 1:3
        corr = abs(xcorr(time_data, pss_time_seq(:, nid2), 'normalized'));
        corr_center = corr(Nfft:end); % Length 256 (lags 0 to 255)
        pss_corr_time(:, nid2) = pss_corr_time(:, nid2) + corr_center(1:Nfft);
    end
end
pss_corr_time = pss_corr_time / Nant; % Average across antennas

% Time-Domain SSS Correlation Over IFFT 256 Samples (Symbol 6)
sss_time_seq = zeros(Nfft, 2); % Subframes 0 and 5
for sf = 1:2
    freq_data = zeros(Nfft, 1);
    freq_data(start_idx:end_idx) = sss_seq(:, N_ID_1_est + 1, sf);
    sss_time_seq(:, sf) = ifft(ifftshift(freq_data), Nfft) * sqrt(Nfft);
end

% Correlate for SSS symbol
sss_corr_time = zeros(Nfft, 2); % 256 x 2
for ant = 1:Nant
    time_data = time_symbols(:, sss_symbol, ant); % 256 samples from symbol 6
    for sf = 1:2
        corr = abs(xcorr(time_data, sss_time_seq(:, sf), 'normalized'));
        corr_center = corr(Nfft:end); % Length 256
        sss_corr_time(:, sf) = sss_corr_time(:, sf) + corr_center(1:Nfft);
    end
end
sss_corr_time = sss_corr_time / Nant; % Average across antennas

% CRS Detection and Channel Estimation
crs_symbols = [1, 4, 8, 11]; % Symbols 0, 3, 7, 10 (1-based)
v1 = mod(PCI_est, 6);
crs_positions = zeros(Nsc, length(crs_symbols));
crs_ref = zeros(Nsc, length(crs_symbols));

% Regenerate CRS
Mpn = 401;
Nc = 1600;
x1 = zeros(1, Mpn + Nc);
x2 = zeros(1, Mpn + Nc);
x1(1) = 1;
x1(2:31) = 0;
for n = 1:Mpn + Nc
    x1(n+31) = mod(x1(n+3) + x1(n), 2);
end
Cinit = 2^10 * (7 * (Nslots + 1) + crs_symbols(1) + 1) * (2 * PCI_est + 1) + 2 * PCI_est + 1;
binStr = dec2bin(Cinit, 31);
bitArray = str2num(binStr(:))';
x2(1:31) = bitArray;
for n = 1:Mpn + Nc
    x2(n+31) = mod(x2(n+3) + x2(n+2) + x2(n+1) + x2(n), 2);
end
C_pseudo = mod(x1(Nc + 1:Nc + Mpn) + x2(Nc + 1:Nc + Mpn), 2);
Reff = zeros(1, 2 * NDLRB);
for m = 1:2 * NDLRB
    Reff(m) = (1/sqrt(2)) * (1 - 2 * C_pseudo(2 * m)) + 1j * (1/sqrt(2)) * (1 - 2 * C_pseudo(2 * m + 1));
end

% Map CRS
for m = 1:2 * NDLRB
    for idx = 1:length(crs_symbols)
        L = crs_symbols(idx);
        if L == 1 || L == 8
            v = 0;
        else
            v = 3;
        end
        k = 6 * m + mod(v + v1, 6);
        if k <= Nsc && k > 0
            crs_positions(k, idx) = 1;
            crs_ref(k, idx) = Reff(m);
        end
    end
end

% Channel estimation
H_est = zeros(Nsc, length(crs_symbols), Nant);
for ant = 1:Nant
    for idx = 1:length(crs_symbols)
        L = crs_symbols(idx);
        for k = 1:Nsc
            if crs_positions(k, idx) == 1
                H_est(k, idx, ant) = rx_grid(k, L, ant) / crs_ref(k, idx);
            end
        end
    end
end

% Interpolate channel estimates
H_est_interp = zeros(Nsc, length(crs_symbols), Nant);
for ant = 1:Nant
    for idx = 1:length(crs_symbols)
        crs_idx = find(crs_positions(:, idx));
        crs_h = H_est(crs_idx, idx, ant);
        all_idx = 1:Nsc;
        H_est_interp(:, idx, ant) = interp1(crs_idx, crs_h, all_idx, 'linear', 'extrap');
    end
end
H_est_final = mean(H_est_interp, 2); % 180 x 1 x 8

% MUSIC Algorithm for AoA Estimation
% Compute spatial covariance matrix
H_avg = mean(H_est_final, 1); % Average over subcarriers: 1 x 1 x 8
H_avg = squeeze(H_avg).'; % 8 x 1
R = H_avg * H_avg'; % Initial covariance matrix
for k = 1:Nsc
    h_k = squeeze(H_est_final(k, 1, :)); % Nant x 1
    R = R + (h_k * h_k'); % Accumulate covariance
end
R = R / Nsc; % Average covariance

% Eigenvalue decomposition
[eig_vec, eig_val] = eig(R);
eig_val = diag(eig_val);
[~, idx] = sort(eig_val, 'descend');
eig_vec = eig_vec(:, idx); % Sort eigenvectors
En = eig_vec(:, Npaths:end); % Noise subspace (8 x (8-Npaths))

% MUSIC spectrum
angles = -90:1:90; % Angle range for scanning
P_music = zeros(size(angles));
for i = 1:length(angles)
    a = exp(-1j * 2 * pi * d_lambda * sind(angles(i)) * (0:Nant-1)).' / sqrt(Nant);
    P_music(i) = 1 / abs(a' * (En * En') * a);
end
P_music = 10 * log10(P_music / max(P_music)); % Normalize and convert to dB

% Find AoA estimates
[~, peak_idx] = findpeaks(P_music, 'SortStr', 'descend', 'NPeaks', Npaths);
theta_est = angles(peak_idx);
theta_est = sort(theta_est); % Sort for consistency
